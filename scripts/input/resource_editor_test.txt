# =============================================================
# Resource Editor — Functional Test Script
# =============================================================
# Run this with:
#   build_ninja\tools\resource_editor\vde_resource_editor.exe scripts\input\resource_editor_test.txt
#
# This exercises all Phase 1 commands. The editor runs in
# script mode — no window interaction required. Watch the
# console output for OK / ERROR on each command.
# =============================================================

# --- Help ---
help
help new
help paint

# --- Canvas management ---
new 32 32 hero
new 16 16 enemy
new 64 64 background
list

# --- Switch active canvas ---
set_active hero

# --- Drawing color / tool / brush size ---
setcolor #FF0000FF
settool brush
setsize 3

# --- Paint some pixels (on active canvas "hero") ---
paint 5 5 #FF0000FF
paint 10 10 #00FF00FF 2
paint 0 0 #0000FFFF

# --- Fill entire canvas ---
fill #222222FF

# --- Flood fill ---
floodfill 0 0 #AAAAAAFF

# --- Shapes ---
line 0 0 31 31 #FFFF00FF
rect 2 2 10 8 #FF00FFFF 1
rect 14 14 6 6 #00FFFFFF 0
circle 16 16 5 #FF8800FF 1
circle 24 24 3 #8800FFFF 0

# --- Undo / Redo ---
undo
undo
redo

# --- Color picker ---
pick 5 5

# --- Transforms ---
fliph
flipv
resize 64 64
crop 0 0 32 32

# --- Zoom / Pan ---
zoom 4
pan 10 10

# --- Target other canvas by name with @prefix ---
@enemy fill #880000FF
@enemy paint 8 8 #FFFFFFFF
@enemy line 0 0 15 15 #00FF00FF

# --- Target by numeric ID (canvas 3 = background) ---
@3 fill #335577FF
@background rect 10 10 20 20 #FFFFFFAA 1

# --- Export / Save ---
@hero export hero_test_output.png
@enemy export enemy_test_output.png

# --- Close a canvas ---
@enemy close
list

# --- Log management ---
log save test_command_log.txt

# --- Exit ---
exit
