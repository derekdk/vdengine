# VDE Resource Editor - Smoke Test Script
# Exercises all commands in sequence for scripted validation.
# Phase 1: 30 commands, Phase 2: +8 commands (named colors, arc/bezier,
# cross-canvas operations, REPL autocomplete).
#
# Run as: vde_resource_editor.exe <path_to_this_file>
# Expected result: exit code 0, no command failures.

# =============================================================================
# Help system
# =============================================================================
help
help create canvas
help fill

# =============================================================================
# Named colors (Phase 2, Step 6)
# =============================================================================

# define color - create named color aliases
define color cherry #FF0044FF
define color sky #4488FFFF
define color grass #22CC22FF

# list colors - display all defined colors
list colors

# Use named colors in drawing (tested after canvas creation below)

# undefine color - remove a named color
undefine color grass

# Verify remaining colors survive
list colors

# =============================================================================
# Canvas management (global commands)
# =============================================================================

# create canvas - create two canvases
create canvas sprite 32 32
create canvas overlay 64 64

# list - show all open canvases
list

# =============================================================================
# Tool configuration (global commands)
# =============================================================================

setcolor #FF0000FF
settool brush
setsize 2

# =============================================================================
# View options (global commands)
# =============================================================================

zoom 2
grid on
grid off
zoom 1

# =============================================================================
# Drawing commands (active canvas: sprite, 32x32)
# =============================================================================

# fill - flood fill entire canvas
fill #FFFFFFFF

# set - set a single pixel
set (5, 5) #FF0000FF

# draw line - Bresenham diagonal
draw line (0, 0) to (31, 31) with #0000FFFF

# draw rect - outline and filled variants
draw rect (2, 2) to (10, 10) with #00FF00FF outline
draw rect (14, 14) to (20, 20) with #FFFF00FF filled

# draw circle - outline and filled variants
draw circle (16, 16) radius 8 with #FF00FFFF outline
draw circle (10, 10) radius 3 with #00FFFFFF filled

# draw ellipse - outline and filled variants
draw ellipse (16, 8) (6, 3) with #FF8800FF outline
draw ellipse (8, 16) (3, 5) with #8800FFFF filled

# flood - flood fill from a corner
flood (0, 31) with #888888FF

# =============================================================================
# Arc and Bezier drawing (Phase 2, Step 9)
# =============================================================================

# draw arc - circular arc, outline
draw arc (16, 16) radius 10 from 0 to 180 with #FF0000FF

# draw arc - with thickness
draw arc (16, 16) radius 6 from 45 to 270 with #00FF00FF width 2

# draw bezier - cubic Bezier curve
draw bezier (0, 0) (10, 0) (20, 31) (31, 31) with #0000FFFF

# draw bezier - with thickness
draw bezier (0, 31) (10, 31) (20, 0) (31, 0) with cherry width 2

# =============================================================================
# Named colors in drawing (Phase 2, Step 6 cont.)
# =============================================================================

# Use a named color for fill
fill sky

# Use a named color in draw line
draw line (0, 0) to (31, 31) with cherry

# =============================================================================
# Undo / redo
# =============================================================================

undo
undo
redo
redo

# =============================================================================
# Transform operations
# =============================================================================

clear
fill #FFFFFFFF
flip horizontal
flip vertical
rotate 90
rotate 270
rotate 180

# =============================================================================
# Resize and crop
# =============================================================================

resize 48 48
crop (0, 0) (32, 32)

# =============================================================================
# History
# =============================================================================

history 10

# =============================================================================
# File operations
# =============================================================================

# Save, then export, then load the saved file as a new canvas
save "smoke_output.png"
export "smoke_output_export.png"
load "smoke_output.png"

# Select the loaded canvas, then delete it
select smoke_output
delete smoke_output

# =============================================================================
# Multi-canvas operations
# =============================================================================

# Switch to overlay and draw on it
select overlay
fill #0000FFFF
draw rect (0, 0) to (63, 63) with #FFFFFFFF outline

# Return to sprite
select sprite

# Rename overlay back and forth to verify rename command
rename overlay renamed_overlay
rename renamed_overlay overlay

# Delete the overlay canvas
delete overlay

# =============================================================================
# Cross-canvas operations (Phase 2, Step 8) — NOT smoke-testable yet
# =============================================================================
# draw image, rehost, and copyhost require sub-resources in a canvas's
# resource map.  There is no command yet to create sub-resources, so these
# commands are tested only via unit tests (Phase2Commands_test.cpp).
# TODO: add smoke coverage once a "create resource" or "import resource"
#       command is available.

# =============================================================================
# Exit
# =============================================================================

# NOTE: do NOT add 'exit' here.  In script mode the engine calls quit()
# automatically after the script finishes.  Calling exit within the script
# also calls game->quit(), which causes a double-quit and an access violation.
